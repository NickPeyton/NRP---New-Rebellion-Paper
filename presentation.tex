    \section{Background}
    \begin{frame}
        \frametitle{Historical Background: The Dissolution of the Monasteries}
        \begin{columns}
            \column{.4\linewidth}
            \includegraphics[height=.99\textwidth, angle=270, origin=c]{Output/Images/dogFountains.jpg}
            \tiny Dirty animals now roam the halls of the once-mighty Fountains Abbey
            \column{.6\linewidth}
        \begin{itemize}
            \item Largest transfer of property in English history since 1066 ($\frac{1}{6}$ to $\frac{1}{20}$ of English agricultural land)
            \item Began in 1536 with the dissolution of all religious houses with an annual income below \pounds200
            \item Large houses dissolved under pressure through 1540
            \item Immediately preceded by \emph{Valor Ecclesiasticus} of 1535, a full survey of all Church land in England
            \item All land taken by Crown
        \end{itemize}
        \end{columns}
    \end{frame}

    
    \begin{frame}
        \frametitle{Historical Background: The Pilgrimage of Grace}
        \begin{columns}
            \column{.6\textwidth}
                \begin{itemize}
                    \item Largest rebellion between 1381 Peasants' Revolt and English Civil War
                    \item Rebellion sparked on 1 October 1536 by Dissolution Commissioners advancing on Louth Park Abbey in Lincolnshire
                    \item Spread rapidly throughout northern England and gathered an army of 30,000 rebels between October and December 1536
                    \item Confronted heavily outnumbered royal force near Doncaster, dispersed by false promises
                \end{itemize}
                
            \column{.4\textwidth}
            \includegraphics[width=.99\linewidth]{Output/Images/pogBanner.png}
            \tiny Banner of the Pilgrims, depicting the Holy Wounds of Christ
        \end{columns}
    \end{frame}

    \begin{frame}
        \frametitle{Historical Background: The Pilgrimage of Grace}
    \begin{columns}
            \column{.6\textwidth}
            \begin{itemize}
                \item Wide-ranging list of grievances:
                \begin{itemize}
                    \item Monasteries
                    \item Liturgy
                    \item Inheritance law
                    \item Taxation
                    \item Commoners on royal council
                \end{itemize}
                \item Historians generally agree that the Pilgrimage was \emph{driven} by the commons but disagree on causes
                \item Parish was the fundamental unit in raising and sustaining rebellion, used pre-existing militia system
                \item Led by gentlemen who claimed they were coerced, unclear how true this was
            \end{itemize}
            \column{.4\textwidth}
            \includegraphics[width=.99\linewidth]{Output/Images/pogBanner.png}
            \tiny Banner of the Pilgrims, depicting the Holy Wounds of Christ

    \begin{frame}
        \frametitle{Why (Just) the North?}
    \begin{columns}
            \column{.6\textwidth}
            Other local disturbances in East Anglia, Oxfordshire, Kent, and Somerset but no full-scale rebellion. Why?
            \begin{enumerate}
            \item Location
            \begin{itemize}
                \item Distant from London
                \item Militarized border
                \item Independent lords
            \end{itemize}
            \item Poverty and Underpopulation
            \begin{itemize}
                \item Closer to subsistence line
                \item Taxes and harvest failures hit harder
                \item Monasteries more important
            \end{itemize}
            \item Monastic Landscape
            \begin{itemize}
                \item 3x regular almsgiving rate
                \item Smaller monasteries
                \item Hospitality and facilitation of grain trade
            \end{itemize}
             \end{enumerate}
            \column{.4\textwidth}
            \includegraphics[width=.99\linewidth]{Output/Images/Maps/taxpayers_map_1524.png}
            \tiny Taxpayer densities of England, 1524 (Sheail 1972)
        \end{columns}
    \end{frame}
    \begin{frame}
        \frametitle{Pilgrimage of Grace: Actors}

        \begin{enumerate}
            \item The Commons

                \item Entry fines, rent hikes, and enclosure
                \item Monastic employment and poor relief
                \item Monastic agricultural demand
                \item Changes to parish religion
            \end{itemize}
            \item The Gentlemen
            \begin{itemize}
                \item Statute of Uses
                \item Personal slights
                \item ``Base-born" ministers
                \item Religious conservatism
            \end{itemize}
            \item The Religious
            \begin{itemize}
                \item Destruction of religious houses
                \item Break with Rome
                \item Doctrinal changes
            \end{itemize}
        \end{enumerate}
    \end{frame}

            \item \textbf{``Musters''} -- gatherings of rebels, initially repeated daily with rebels \textbf{going home at night}

    \section{Methods and Results}
    \begin{frame}
        \frametitle{Methods: Parish-Level Regressions}
\begin{math}
    \begin{aligned}
Rebellion_{i} = \beta_{0} & + \beta_{1}MonasticLand_{i} + \beta_{2}Tithe + \beta_{3}Alms \\ 
& + \beta_{4}MonasticIncome + \beta_{5}TaxLevel_{i} \\
& + \beta_{6}Wet1535_{i} +\beta_{7}Wet1536_{i} \\
& + \beta_{8}Percy_{i} + \beta_{9}Population_{i} \\
& + \beta_{10}\textbf{X}_{i} + \epsilon_i
    \end{aligned}
\end{math}

        \begin{itemize}
            \item Rebellion = one of 3 rebellion variables: primary muster dummy, muster dummy, or number of seats of rebel gentlemen
            \item ``Wet" = PDSI for grid cell that year, Percy = dummy for ``within 20km of a Percy country seat"
            \item Logistic regression used for musters, Poisson for rebel gentlemen, Cox PH for survival analysis
        \end{itemize}
    \end{frame}

    \section{Discussion}
    \begin{frame}
        \frametitle{Discussion: Key Variables}
        \begin{enumerate}
            \item Monastic Land
            \begin{itemize}
                \item Monastic land consistently predicts rebellion
                \item Land of large monasteries and off-site land predicts rebellion more strongly
            \end{itemize}
            \item Wet 1536
            \begin{itemize}
                \item Second poor harvest in a row
                \item Subsistence crises tend to lead to revolt
            \end{itemize}
            \item Percy
            \begin{itemize}
                \item Resentment against Royal pressure on estates
                \item Deep and wide network in the North
            \end{itemize}
        \end{enumerate}
    \end{frame}
    \begin{frame}
        \frametitle{Discussion: Land}
        Why does (large-house, off-site) monastic land predict rebellion?
        \begin{enumerate}
            \item Fear of changes in tenancy: inflation and rise of entry fines during 16th century
            \item Monastic recruiting shifted toward tenants and their neighbors in the century before 1536, monks did favors for family and friends
            \item Monasteries were a large and reliable source of demand for their tenants' agricultural products
            \item Median pound of land income for Northern monasteries came from less than 8km away
            \item Monks of some large abbeys (Furness, Holmcultram) pushed tenants to rebel
        \end{enumerate}
    \end{frame}

    \begin{frame}
        \frametitle{Discussion: Musters vs. Rebel Gentlemen}
        Results may indicate a difference in motivations between gentlemen and commons:
        \begin{itemize}
            \item Taxation level predicts rebel gentlemen in final specification
            \item Level of taxation does not provide information about the \emph{rate} of taxation
            \item Monastic land predicts rebel gentlemen, likely trades off directly with presence of \emph{any} gentleman's seat, so potentially underestimated
            \item No map of non-rebel gentlemen, so high taxation per taxpayer may just reflect the presence of \emph{any} gentleman
        \end{itemize}
        
    \end{frame}

    \section{Conclusion}
    \begin{frame}
        \frametitle{Conclusion}
        \begin{itemize}
            \item Monastic land consistently predicts rebel activity
            \item Off-site (more-tenanted) and larger monastery land predicts rebel activity even more strongly
            \item Taxation levels do not predict rebel activity
            \item Key role for wet weather $\rightarrow$ poor harvest $\rightarrow$ rebellion and Percy influence
            \item More work needed on demesne vs non-demesne land, rebel vs non-rebel gentlemen, role of Crown-influenced abbots
            \item \textbf{Bush was right}
            \item Future plans: more data on non-rebel gentlemen, evidence on Royal pressure in monastic elections, names of monastic patrons
        \end{itemize}
        
    \end{frame}
